cmake_minimum_required(VERSION 3.22.1)
project(fourthward_xr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenXR REQUIRED CONFIG)
add_library(native_app_glue STATIC ${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c)
target_include_directories(native_app_glue PUBLIC ${ANDROID_NDK}/sources/android/native_app_glue)
add_library(fourthward_xr SHARED openxr_main.cpp)
target_compile_definitions(fourthward_xr PRIVATE XR_USE_PLATFORM_ANDROID XR_USE_GRAPHICS_API_OPENGL_ES)
target_compile_options(fourthward_xr PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
target_link_libraries(fourthward_xr PRIVATE OpenXR::openxr_loader native_app_glue android log EGL GLESv3 jnigraphics)
target_link_options(fourthward_xr PRIVATE -Wl,-u,ANativeActivity_onCreate)
